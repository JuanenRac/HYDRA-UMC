# Single-camera skeleton (verify TAPPAS element names/properties against the
# actually-installed TAPPAS version before trusting this syntax - not run
# against real hardware):
gst-launch-1.0 \
  v4l2src device=/dev/video0 ! \
  video/x-raw,format=YUY2,width=1280,height=720,framerate=30/1 ! \
  videoconvert ! \
  hailonet hef-path=../models/yolov8_hydra.hef ! \
  hailofilter so-path=libyolo_post.so ! \
  hailooverlay ! \
  videoconvert ! \
  x264enc tune=zerolatency ! \
  rtph264pay ! \
  udpsink host=127.0.0.1 port=5000
